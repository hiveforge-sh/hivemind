   ```typescript
   private async loadGraphData(): Promise<GraphData> {
     // Get active file for local graph mode
     const activeFile = this.plugin.app.workspace.getActiveFile();
     const viewMode = this.plugin.settings.graphViewMode;

     if (viewMode === 'local' && activeFile) {
       // Local graph: get neighbors of current entity
       const entityName = activeFile.basename;

       try {
         const response = await this.plugin.callMCPTool('hvmd_graph_get_neighbors', {
           entity_id: entityName,
           depth: 1,
           direction: 'both'
         }) as MCPResponse;

         const result = JSON.parse(response.content[0].text);
         return this.transformNeighborsResponse(result, entityName, activeFile.path);
       } catch (error) {
         // Entity might not exist, fallback to subgraph
         console.log('Local graph fallback to subgraph');
       }
     }

     // Full graph: get subgraph from first available entity
     // Use hvmd_graph_get_subgraph with depth 2 for broader view
     const response = await this.plugin.callMCPTool('hvmd_list_entities', {
       limit: 1
     }) as MCPResponse;

     const entities = JSON.parse(response.content[0].text);
     if (entities.entities && entities.entities.length > 0) {
       const firstEntity = entities.entities[0];
       const subgraphResponse = await this.plugin.callMCPTool('hvmd_graph_get_subgraph', {
         entity_id: `${firstEntity.type}:${firstEntity.title}`,
         depth: 2,
         direction: 'both'
       }) as MCPResponse;

       const subgraph = JSON.parse(subgraphResponse.content[0].text);
       return this.transformSubgraphResponse(subgraph);
     }

     return { nodes: [], edges: [] };
   }

   private transformNeighborsResponse(result: any, centerName: string, centerPath: string): GraphData {
     const nodes: GraphNode[] = [];
     const edges: GraphEdge[] = [];
     const seenNodes = new Set<string>();

     // Add center node
     const centerId = `center:${centerName}`;
     nodes.push({
       id: centerId,
       title: centerName,
       type: result.source?.type || 'unknown',
       path: centerPath
     });
     seenNodes.add(centerId);

     // Add neighbor nodes and edges
     if (result.neighbors) {
       result.neighbors.forEach((neighbor: any) => {
         const neighborId = neighbor.node?.id || neighbor.id || '';
         if (!seenNodes.has(neighborId)) {
           nodes.push({
             id: neighborId,
             title: neighbor.node?.title || neighbor.title || neighborId,
             type: neighbor.node?.type || neighbor.type || 'unknown',
             path: neighbor.node?.path || neighbor.path || ''
           });
           seenNodes.add(neighborId);
         }

         edges.push({
           source: neighbor.relationship?.direction === 'incoming' ? neighborId : centerId,
           target: neighbor.relationship?.direction === 'incoming' ? centerId : neighborId,
           relationshipType: neighbor.relationship?.type || 'related'
         });
       });
     }

     return { nodes, edges };
   }

   private transformSubgraphResponse(result: any): GraphData {
     const nodes: GraphNode[] = [];
     const edges: GraphEdge[] = [];
     const seenNodes = new Set<string>();

     // Add all nodes from subgraph
     if (result.nodes) {
       result.nodes.forEach((node: any) => {
         const nodeId = node.id || `${node.type}:${node.title}`;
         if (!seenNodes.has(nodeId)) {
           nodes.push({
             id: nodeId,
             title: node.title || nodeId,
             type: node.type || 'unknown',
             path: node.path || ''
           });
           seenNodes.add(nodeId);
         }
       });
     }

     // Add all edges
     if (result.edges) {
       result.edges.forEach((edge: any) => {
         edges.push({
           source: edge.source || edge.from,
           target: edge.target || edge.to,
           relationshipType: edge.relationshipType || edge.type || 'related'
         });
       });
     }

     return { nodes, edges };
   }
   ```

   ```typescript
   async onOpen() {
     const container = this.containerEl.children[1] as HTMLElement;
     container.empty();
     container.addClass('hivemind-graph-view');

     // Show loading state
     const loadingEl = container.createDiv({ cls: 'hvmd-scanning' });
     loadingEl.setText('Loading graph data...');

     try {
       const data = await this.loadGraphData();

       if (data.nodes.length === 0) {
         container.empty();
         container.createDiv({
           cls: 'hvmd-no-issues',
           text: 'No entities with relationships found. Create relationships in your frontmatter.'
         });
         return;
       }

       // Clear loading
       container.empty();
       container.addClass('hivemind-graph-view');

       // Build type count map for toolbar
       const typeCountMap = new Map<string, number>();
       data.nodes.forEach(node => {
         const count = typeCountMap.get(node.type) || 0;
         typeCountMap.set(node.type, count + 1);
       });

       // Create toolbar with view mode toggle (Local/Full)
       this.createToolbar(container, typeCountMap);

       // Create sigma container
       const graphContainer = container.createDiv({
         cls: 'hvmd-graph-container'
       });
       graphContainer.style.height = 'calc(100% - 80px)';  // Account for toolbar

       // Create graphology instance
       this.graph = new Graph();
       data.nodes.forEach(node => {
         this.graph!.addNode(node.id, {
           label: node.title,
           type: node.type,
           path: node.path,
           size: 10,  // Base node size
           x: Math.random() * 100,  // Random initial positions
           y: Math.random() * 100
         });
       });

       data.edges.forEach(edge => {
         // Avoid duplicate edges
         if (!this.graph!.hasEdge(edge.source, edge.target)) {
           try {
             this.graph!.addEdge(edge.source, edge.target, {
               label: edge.relationshipType,
               type: edge.relationshipType
             });
           } catch (e) {
             // Skip edges with missing nodes
             console.log('Skipping edge with missing node:', edge);
           }
         }
       });

       // Apply ForceAtlas2 layout
       // Per research: use barnesHutOptimize for >1000 nodes
       const nodeCount = this.graph.order;
       forceAtlas2.assign(this.graph, {
         iterations: 50,
         settings: {
           gravity: 1,
           scalingRatio: 10,
           slowDown: 1,
           barnesHutOptimize: nodeCount > 1000,
           barnesHutTheta: 0.5,
           linLogMode: false,
           strongGravityMode: false,
           edgeWeightInfluence: 1
         }
       });

       // Warn on large graphs per CONTEXT.md
       if (nodeCount > 100) {
         new Notice(`Large graph (${nodeCount} nodes). Consider filtering by entity type.`);
       }

       // Create sigma renderer with Okabe-Ito coloring
       this.renderer = new Sigma(this.graph, graphContainer, {
         renderEdgeLabels: false,  // Show on hover only per CONTEXT.md
         enableEdgeHoverEvents: true,
         enableEdgeClickEvents: true,
         labelSize: 12,
         labelWeight: 'bold',
         nodeReducer: this.getNodeReducer(),
         edgeReducer: this.getEdgeReducer()
       });

       // Fit camera to show all nodes
       this.renderer.getCamera().animatedReset({ duration: 0 });

     } catch (error) {
       container.empty();
       const errorEl = container.createDiv({ cls: 'hvmd-no-issues' });
       const errorMsg = (error as Error).message;

       errorEl.createEl('p', {
         text: `Failed to load graph: ${errorMsg}`,
         cls: 'hvmd-text-warning'
       });

       // If error is about MCP connection, show connect button
       if (errorMsg.includes('MCP server not connected')) {
         const connectBtn = errorEl.createEl('button', {
           text: 'Connect to MCP',
           cls: 'mod-cta'
         });
         connectBtn.addEventListener('click', () => {
           void this.plugin.startMCPServer();
           setTimeout(() => void this.onOpen(), 2000);
         });
       } else {
         const retryBtn = errorEl.createEl('button', { text: 'Retry' });
         retryBtn.addEventListener('click', () => void this.onOpen());
       }
     }
   }
   ```

   ```typescript
   private getNodeReducer() {
     // Okabe-Ito color palette per Phase 26 and CONTEXT.md
     const colorMap: Record<string, string> = {
       character: '#56B4E9',  // Sky blue
       location: '#009E73',   // Green
       event: '#E69F00',      // Orange
       faction: '#F0E442',    // Yellow
       item: '#0072B2',       // Dark blue
       concept: '#D55E00',    // Red-orange
       timeline: '#CC79A7',   // Pink
       default: '#888888'     // Gray fallback
     };

     return (node: string, data: Record<string, unknown>) => {
       const res = { ...data } as Record<string, unknown>;
       const nodeType = (data.type as string) || 'default';
       res.color = colorMap[nodeType] || colorMap.default;
       return res;
     };
   }
   ```
