   private async findShortestPath(sourceId: string, targetId: string) {
     if (!this.graph) return;

     try {
       // Use bidirectional Dijkstra for shortest path
       const path = bidirectional(this.graph, sourceId, targetId);

       if (!path || path.length === 0) {
         new Notice('No path found between selected nodes');
         return;
       }

       // Store path nodes for highlighting
       this.pathNodes = new Set(path);
       this.pathEdges = new Set();

       // Get edges in path
       for (let i = 0; i < path.length - 1; i++) {
         const edge = this.graph.edge(path[i], path[i + 1]);
         if (edge) {
           this.pathEdges.add(edge);
         }
         // Also check reverse direction for undirected edges
         const reverseEdge = this.graph.edge(path[i + 1], path[i]);
         if (reverseEdge) {
           this.pathEdges.add(reverseEdge);
         }
       }

       // Refresh to apply highlighting
       this.renderer?.refresh();

       // Build path description
       const pathLabels = path.map(nodeId =>
         this.graph?.getNodeAttribute(nodeId, 'label') as string || nodeId
       );
       new Notice(`Path: ${pathLabels.join(' → ')} (${path.length} nodes)`);

     } catch (error) {
       new Notice(`Error finding path: ${(error as Error).message}`);
     }
   }
   ```

     if (!this.graph) return;

     try {
       // Run Louvain community detection
       const result = louvain(this.graph, { resolution: 1.0 });

       // Store community assignments
       this.communities = new Map(Object.entries(result));

       // Count communities
       const uniqueCommunities = new Set(Object.values(result));
       new Notice(`Detected ${uniqueCommunities.size} communities`);

       // Enable cluster coloring
       this.showClusters = true;
       this.plugin.settings.graphShowClusters = true;
       void this.plugin.saveSettings();

       // Refresh to apply cluster colors
       this.renderer?.refresh();

   private async saveLayout() {
     if (!this.graph) return;

     const positions: Record<string, { x: number; y: number }> = {};
     this.graph.forEachNode((node, attributes) => {
       positions[node] = {
         x: attributes.x as number,
         y: attributes.y as number
       };
     });

     this.plugin.settings.graphLayoutPositions = positions;
     await this.plugin.saveSettings();
     new Notice(`Layout saved (${Object.keys(positions).length} nodes)`);
   }

   private async restoreLayout() {
     if (!this.graph) return;

     const positions = this.plugin.settings.graphLayoutPositions;
     if (!positions || Object.keys(positions).length === 0) {
       new Notice('No saved layout found');
       return;
     }

     let restored = 0;
     this.graph.forEachNode((node) => {
       if (positions[node]) {
         this.graph!.setNodeAttribute(node, 'x', positions[node].x);
         this.graph!.setNodeAttribute(node, 'y', positions[node].y);
         restored++;
       }
     });

     this.renderer?.refresh();
     new Notice(`Restored layout for ${restored} nodes`);
   }
   ```
