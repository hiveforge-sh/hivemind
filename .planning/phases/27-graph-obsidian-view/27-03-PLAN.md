   ```typescript
   private async expandNode(nodeId: string) {
     // Get node title for MCP query
     const nodeTitle = this.graph?.getNodeAttribute(nodeId, 'label') as string;
     const nodeType = this.graph?.getNodeAttribute(nodeId, 'type') as string;

     try {
       const response = await this.plugin.callMCPTool('hvmd_graph_get_neighbors', {
         entity_id: nodeType ? `${nodeType}:${nodeTitle}` : nodeTitle,
         depth: 1,
         direction: 'both'
       }) as MCPResponse;

       const result = JSON.parse(response.content[0].text);

       // Add new nodes and edges
       if (result.neighbors) {
         result.neighbors.forEach((neighbor: any) => {
           const neighborId = neighbor.node?.id || `${neighbor.node?.type}:${neighbor.node?.title}`;

           // Add node if not exists
           if (!this.graph!.hasNode(neighborId)) {
             const existingPos = this.graph!.getNodeAttribute(nodeId, 'x');
             this.graph!.addNode(neighborId, {
               label: neighbor.node?.title || neighborId,
               type: neighbor.node?.type || 'unknown',
               path: neighbor.node?.path || '',
               size: 10,
               x: (existingPos as number) + (Math.random() - 0.5) * 50,
               y: (this.graph!.getNodeAttribute(nodeId, 'y') as number) + (Math.random() - 0.5) * 50
             });
           }

           // Add edge if not exists
           const source = neighbor.relationship?.direction === 'incoming' ? neighborId : nodeId;
           const target = neighbor.relationship?.direction === 'incoming' ? nodeId : neighborId;
           if (!this.graph!.hasEdge(source, target)) {
             this.graph!.addEdge(source, target, {
               label: neighbor.relationship?.type || 'related',
               type: neighbor.relationship?.type || 'related'
             });
           }
         });

         // Re-run layout on new nodes only (incremental)
         forceAtlas2.assign(this.graph!, {
           iterations: 20,
           settings: {
             gravity: 1,
             scalingRatio: 10,
             barnesHutOptimize: this.graph!.order > 1000
           }
         });

         this.renderer?.refresh();
         new Notice(`Expanded ${result.neighbors.length} neighbors`);
       }
     } catch (error) {
       new Notice(`Failed to expand: ${(error as Error).message}`);
     }
   }
   ```

   ```typescript
     const nodeTitle = this.graph?.getNodeAttribute(nodeId, 'label') as string;
     const nodeType = this.graph?.getNodeAttribute(nodeId, 'type') as string;

     // Store focus target and switch to local mode centered on this node

     await this.plugin.saveSettings();

     // Find and open the note to make it "active"
     const path = this.graph?.getNodeAttribute(nodeId, 'path') as string;
     if (path) {
       const file = this.plugin.app.vault.getAbstractFileByPath(path);
       if (file instanceof TFile) {
         // Open file in background to make it the "active" file
         await this.plugin.app.workspace.getLeaf('tab').openFile(file);
         // Reload graph view with new active file as center
         void this.onOpen();
       }
     } else {
       new Notice('Cannot focus: node has no associated file');
     }
   }
   ```
