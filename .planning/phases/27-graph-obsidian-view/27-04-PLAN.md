5. Add filter state persistence methods:
   ```typescript
   private loadFilterState(availableTypes: string[]) {
     const savedTypes = this.plugin.settings.graphFilterTypes;
     if (savedTypes && savedTypes.length > 0) {
       this.activeTypeFilters = new Set(savedTypes.filter(t => availableTypes.includes(t)));
       if (this.activeTypeFilters.size === 0) {
         this.activeTypeFilters = new Set(availableTypes);
       }
     } else {
       this.activeTypeFilters = new Set(availableTypes);
     }

     const savedRelationships = this.plugin.settings.graphFilterRelationships;
     if (savedRelationships && savedRelationships.length > 0) {
       this.activeRelationshipFilters = new Set(savedRelationships);
     }
   }

   private async saveFilterState() {
     this.plugin.settings.graphFilterTypes = Array.from(this.activeTypeFilters);
     this.plugin.settings.graphFilterRelationships = Array.from(this.activeRelationshipFilters);
     await this.plugin.saveSettings();
   }
   ```

   ```typescript
       character: '#56B4E9',
       location: '#009E73',
       event: '#E69F00',
       faction: '#F0E442',
       item: '#0072B2',

       timeline: '#CC79A7',
       default: '#888888'
     };

     return (node: string, data: Record<string, unknown>) => {
       const res = { ...data } as Record<string, unknown>;
       const nodeType = (data.type as string) || 'default';
       res.color = colorMap[nodeType] || colorMap.default;

       // GVIEW-06: Filter by entity type
       if (!this.activeTypeFilters.has(nodeType)) {
         res.hidden = true;
       }

       // GVIEW-07: Highlight search results
       if (this.highlightedNodes.has(node)) {
         res.color = '#CC79A7';  // Pink highlight
         res.size = (data.size as number || 10) * 1.5;  // Larger for visibility
       }

       return res;
     };
   }
   ```

   ```typescript
   const searchSection = toolbar.createDiv({ cls: 'hvmd-graph-toolbar-section' });
   searchSection.createEl('span', { text: 'Search: ', cls: 'hvmd-graph-label' });

   });
   searchInput.value = this.searchQuery;

   // Debounced search
   let searchTimeout: NodeJS.Timeout | null = null;
   searchInput.addEventListener('input', () => {
     if (searchTimeout) clearTimeout(searchTimeout);
     searchTimeout = setTimeout(() => {
       this.searchQuery = searchInput.value;
       this.performSearch(searchInput.value);
     }, 200);  // 200ms debounce
   });

   // Clear button
   const clearBtn = searchSection.createEl('button', {
     text: 'Clear',
     cls: 'hvmd-graph-btn hvmd-graph-btn-small'
   });
   clearBtn.addEventListener('click', () => {
     searchInput.value = '';
     this.searchQuery = '';
     this.highlightedNodes.clear();
     this.renderer?.refresh();
   });
   ```

2. Add performSearch method:
   ```typescript
   private performSearch(query: string) {
     if (!this.graph || !this.renderer) return;

     this.highlightedNodes.clear();

     if (!query.trim()) {
       this.renderer.refresh();
       return;
     }

     const lowerQuery = query.toLowerCase();
     const matchingNodes: string[] = [];

     this.graph.forEachNode((node, attributes) => {
       const label = (attributes.label as string || '').toLowerCase();
       if (label.includes(lowerQuery)) {
         matchingNodes.push(node);
         this.highlightedNodes.add(node);
       }
     });

     if (matchingNodes.length === 0) {
       new Notice('No matching nodes found');
       return;
     }

     new Notice(`Found ${matchingNodes.length} matching nodes`);

     // Refresh to apply highlighting
     this.renderer.refresh();

     // Pan camera to first match
     if (matchingNodes.length > 0) {
       const firstNode = matchingNodes[0];
       const camera = this.renderer.getCamera();
       const nodePos = {
         x: this.graph.getNodeAttribute(firstNode, 'x') as number,
         y: this.graph.getNodeAttribute(firstNode, 'y') as number
       };

       // Animate camera to first match
       camera.animate(nodePos, { duration: 500 });
     }
   }
   ```
